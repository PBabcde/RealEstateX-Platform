Dat is een cruciale integratietest voor de veiligheid en data-integriteit van het systeem! Het testen van de opruimroutine tegen een echte PostgreSQL-instantie (in plaats van alleen unit-mocks) geeft 100% zekerheid dat er nooit actieve sessies of klantdata verloren kunnen gaan.
Hier is de specificatie en uitwerking voor Taak #77:
Opdracht & Specificatie: Taak #77
Doel: Garanderen dat het opruimscript voor oude sessies (legacy-session-cleanup.ts) op een echte PostgreSQL-database:
 * Alleen verlopen sessies buiten de gratieperiode van 30 dagen verwijdert.
 * Actieve sessies, sessies binnen de 30 dagen gratieperiode, users-records, audit-logs en overige applicatiedata volledig intact laat.
 * Bij een mismatch tussen het verwachte (preview) aantal en het daadwerkelijke aantal de gehele transactie rollbackt zonder gegevens te wissen.
1. Integration Test Fixture (tests/legacy_session_cleanup.test.js)
Opzetten van de disposable PostgreSQL-testsuite met randgevallen rondom de vervaldatums en transactie-rollback:
// tests/legacy_session_cleanup.test.js
import { describe, it, before, after, beforeEach } from 'node:test';
import assert from 'node:assert/strict';
import { pgClient, setupSchema, teardownSchema } from './helpers/pg-fixture.js';
import { runLegacySessionCleanup } from '../scripts/src/legacy-session-cleanup.js';

describe('Taak #77: Legacy Session Cleanup PostgreSQL Integration Test', () => {
  before(async () => {
    await setupSchema(); // Laadt lib/db/src/schema/auth.ts en index.ts
  });

  after(async () => {
    await teardownSchema();
  });

  beforeEach(async () => {
    await pgClient.query('TRUNCATE TABLE users, sessions, audit_logs CASCADE;');
  });

  it('behoudt actieve sessies en sessies binnen 30 dagen grace period, verwijdert alleen oudere rows', async () => {
    const now = new Date();
    const daysAgo = (days) => new Date(now.getTime() - days * 24 * 60 * 60 * 1000);

    // Seed test-gebruiker
    const userRes = await pgClient.query(
      "INSERT INTO users (name, email) VALUES ('Eli', 'test@example.com') RETURNING id;"
    );
    const userId = userRes.rows[0].id;

    // Seed diverse sessie-statussen
    await pgClient.query(`
      INSERT INTO sessions (id, user_id, expires_at) VALUES
        ('active_session', '${userId}', '${daysAgo(-1).toISOString()}'),  -- Actief
        ('recent_expired', '${userId}', '${daysAgo(15).toISOString()}'), -- Verlopen < 30 dagen (grace)
        ('old_expired_1',  '${userId}', '${daysAgo(35).toISOString()}'), -- Verlopen > 30 dagen (kandidaat)
        ('old_expired_2',  '${userId}', '${daysAgo(60).toISOString()}'); -- Verlopen > 30 dagen (kandidaat)
    `);

    // Voer opruimscript uit met correcte verwachte count (2 rows)
    const result = await runLegacySessionCleanup({ pgClient, expectedDeleteCount: 2 });

    assert.equal(result.deletedCount, 2);

    // Verifieer welke sessies er over zijn gebleven
    const remainingSessions = await pgClient.query('SELECT id FROM sessions ORDER BY id ASC;');
    const remainingIds = remainingSessions.rows.map((r) => r.id);

    assert.deepEqual(remainingIds, ['active_session', 'recent_expired']);

    // Verifieer dat user record en audit logs ongewijzigd zijn
    const remainingUsers = await pgClient.query('SELECT id FROM users;');
    assert.equal(remainingUsers.rows.length, 1);
  });

  it('rollbackt de volledige transactie als de preview-count niet overeenkomt', async () => {
    const now = new Date();
    const daysAgo = (days) => new Date(now.getTime() - days * 24 * 60 * 60 * 1000);

    const userRes = await pgClient.query(
      "INSERT INTO users (name, email) VALUES ('Eli', 'test@example.com') RETURNING id;"
    );
    const userId = userRes.rows[0].id;

    await pgClient.query(`
      INSERT INTO sessions (id, user_id, expires_at) VALUES
        ('old_expired_1', '${userId}', '${daysAgo(40).toISOString()}'),
        ('old_expired_2', '${userId}', '${daysAgo(50).toISOString()}');
    `);

    // Geef een verkeerde expectedDeleteCount op (bijv. 99 in plaats van 2)
    await assert.rejects(
      async () => {
        await runLegacySessionCleanup({ pgClient, expectedDeleteCount: 99 });
      },
      /preview count mismatch/i
    );

    // Verifieer dat er dankzij de rollback NIETS is verwijderd
    const countRes = await pgClient.query('SELECT COUNT(*) FROM sessions;');
    assert.equal(parseInt(countRes.rows[0].count, 10), 2);
  });
});
