To address Issue #57 (Keep project data together if browser deletion fails), you need to verify and test the atomicity of the deletion flow using IndexedDB transactions. Because IndexedDB transactions automatically roll back all mutations if any error occurs before commit, failure during deletion should keep both the project record and its associated photo records intact.
Here is the implementation and testing strategy across your relevant files:
1. Verification of Transaction Atomicity in DB Module
Ensure that project deletion and photo record removal occur within the exact same transaction block using 'readwrite'. If an error occurs, the transaction aborts and all operations roll back automatically.
artifacts/realestatex/src/lib/property-transformer-db.ts
export async function deleteProject(projectId: string): Promise<void> {
  const db = await getDBInstance();
  
  return new Promise((resolve, reject) => {
    // Open a single readwrite transaction spanning both object stores
    const tx = db.transaction(['projects', 'photos'], 'readwrite');
    const projectStore = tx.objectStore('projects');
    const photoStore = tx.objectStore('photos');

    tx.onerror = () => {
      // Transaction rollback is handled by IndexedDB on error
      reject(tx.error || new Error(`Failed to delete project ${projectId}`));
    };

    tx.oncomplete = () => {
      resolve();
    };

    // 1. Delete associated photo records (or photo index references)
    const photoIndex = photoStore.index('projectId');
    const photoReq = photoIndex.openKeyCursor(IDBKeyRange.only(projectId));
    
    photoReq.onsuccess = (event) => {
      const cursor = (event.target as IDBRequest<IDBCursor>).result;
      if (cursor) {
        photoStore.delete(cursor.primaryKey);
        cursor.continue();
      } else {
        // 2. Delete the primary project record once photos are queued
        projectStore.delete(projectId);
      }
    };
  });
}

2. UI Error Handling & Persistence
Ensure that if deleteProject rejects, the UI catches the error, displays a notification/alert to the user, and keeps the project item visible in state.

const handleDeleteProject = async (projectId: string) => {
  try {
    await deleteProject(projectId);
    // Remove from UI state only upon successful transaction commit
    setProjects((prev) => prev.filter((p) => p.id !== projectId));
    toast.success("Project and associated photos deleted successfully.");
  } catch (error) {
    console.error("Deletion failed:", error);
    // Preserves the UI state and notifies the user
    toast.error("Failed to delete project. Your data was not modified.");
  }
};

3. Browser Test Coverage for Transaction Failure
Simulate a forced transaction abort (e.g., triggering tx.abort() or throwing inside a request event) and assert that both the project and photo records remain completely intact in storage.

import { test, expect } from '@playwright/test'; // or your browser runner framework

test('Forced deletion transaction failure preserves project and photo records', async ({ page }) => {
  await page.goto('/property-transformer');

  // Setup: Inject test project and linked photo records into IndexedDB
  const mockProjectId = 'test-project-123';
  await page.evaluate(async (id) => {
    const db = await window.getDBInstance();
    const tx = db.transaction(['projects', 'photos'], 'readwrite');
    await tx.objectStore('projects').put({ id, name: 'Test Property' });
    await tx.objectStore('photos').put({ id: 'photo-1', projectId: id, url: 'blob:...' });
    await new Promise((res) => (tx.oncomplete = res));
  }, mockProjectId);

  // Act: Attempt deletion with a forced transaction failure injected
  const errorHandled = await page.evaluate(async (id) => {
    const db = await window.getDBInstance();
    
    try {
      await new Promise((resolve, reject) => {
        const tx = db.transaction(['projects', 'photos'], 'readwrite');
        tx.onerror = () => reject(new Error('Transaction aborted'));
        
        // Delete photo
        tx.objectStore('photos').delete('photo-1');
        
        // Force an intentional failure mid-transaction
        tx.abort(); 
      });
      return false;
    } catch (err) {
      return true; // Caught expected failure
    }
  }, mockProjectId);

  expect(errorHandled).toBe(true);

  // Assert 1: Verify saved project and referenced photos still exist in IndexedDB
  const storedData = await page.evaluate(async (id) => {
    const db = await window.getDBInstance();
    const tx = db.transaction(['projects', 'photos'], 'readonly');
    const project = await tx.objectStore('projects').get(id);
    const photo = await tx.objectStore('photos').get('photo-1');
    return { project, photo };
  }, mockProjectId);

  expect(storedData.project).toBeDefined();
  expect(storedData.project.id).toBe(mockProjectId);
  expect(storedData.photo).toBeDefined();

  // Assert 2: UI error toast/message is displayed and project remains in list
  await expect(page.locator(`[data-project-id="${mockProjectId}"]`)).toBeVisible();
  await expect(page.locator('.toast-error')).toContainText('Failed to delete project');
});
