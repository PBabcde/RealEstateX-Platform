Create utility functions to test whether localStorage is accessible and handle write errors gracefully.
// artifacts/realestatex/src/lib/property-transformer-db.ts

const PORTFOLIO_ORDER_KEY = 'portfolio_project_order';

export interface StorageResult<T> {
  data: T | null;
  isStorageAvailable: boolean;
}

/**
 * Safely reads the saved project order preference.
 */
export function getSavedProjectOrder<T>(defaultOrder: T): StorageResult<T> {
  try {
    const saved = localStorage.getItem(PORTFOLIO_ORDER_KEY);
    return {
      data: saved ? JSON.parse(saved) : defaultOrder,
      isStorageAvailable: true,
    };
  } catch (error) {
    return {
      data: defaultOrder,
      isStorageAvailable: false,
    };
  }
}

/**
 * Safely attempts to persist project order.
 * Returns boolean indicating whether storage succeeded.
 */
export function saveProjectOrder<T>(order: T): boolean {
  try {
    localStorage.setItem(PORTFOLIO_ORDER_KEY, JSON.stringify(order));
    return true;
  } catch (error) {
    console.warn('Unable to persist project order to localStorage:', error);
    return false;
  }
}

import { test, expect } from '@playwright/test'; // or your test runner framework

test.describe('Project Order Storage Fallback', () => {
  test('allows sorting in-session and shows warning when localStorage write fails', async ({ page }) => {
    await page.goto('/property-transformer');

    // Mock localStorage.setItem to throw an SecurityError/QuotaExceededError
    await page.evaluate(() => {
      Storage.prototype.setItem = () => {
        throw new DOMException('The operation is insecure.', 'SecurityError');
      };
    });

    // Re-order the project list via UI
    await page.click('[data-testid="sort-by-date"]');

    // Verify warning notice is visible
    const warning = page.locator('[role="status"]');
    await expect(warning).toBeVisible();
    await expect(warning).toContainText('prevented saving it for future visits');

    // Verify the list still sorted in the current DOM
    const firstProject = page.locator('.project-card').first();
    await expect(firstProject).toContainText('Expected First Item');
  });

  test('handles initial blocked localStorage read gracefully', async ({ page }) => {
    // Mock localStorage.getItem to throw prior to page load
    await page.addInitScript(() => {
      Storage.prototype.getItem = () => {
        throw new DOMException('Access denied', 'SecurityError');
      };
    });

    await page.goto('/property-transformer');

    // Warning should show on load if initial read fails
    const warning = page.locator('[role="status"]');
    await expect(warning).toBeVisible();
  });
});
