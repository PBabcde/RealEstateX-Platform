VALUATION_MODULE = """
class ValuationX:
    '''
    Woningwaardering:
    - WOZ-schatting
    - marktwaarde
    - vergelijkingsobjecten
    '''

    def calculate(self, property_data: dict) -> dict:
        base = property_data.get("woonoppervlak", 0) * 3000
        modifier = 0

        if property_data.get("energielabel") in ["A", "B"]:
            modifier += 15000

        return {
            "status": "ok",
            "valuation": base + modifier
        }
"""

RENTAL_MODULE = """
class RentalX:
    '''
    Huurmodule:
    - huurprijs
    - contracten
    - rendement
    '''

    def rental_overview(self, data: dict) -> dict:
        price = data.get("huurprijs", 0)
        costs = data.get("kosten", 0)
        rendement = price * 12 - costs

        return {
            "status": "ok",
            "annual_rendement": rendement
        }
"""

MORTGAGE_MODULE = """
class MortgageX:
    '''
    Hypotheekmodule:
    - rente
    - aflossing
    - maandlasten
    '''

    def calculate(self, principal: float, interest: float, years: int) -> dict:
        monthly_rate = interest / 12
        months = years * 12
        payment = (principal * monthly_rate) / (1 - (1 + monthly_rate) ** -months)

        return {
            "status": "ok",
            "monthly_payment": round(payment, 2)
        }
"""

STORAGE_MODULE = """
import json
import os

class RealEstateStorageX:
    BASE_PATH = "storage/realestateX/"

    def __init__(self):
        os.makedirs(self.BASE_PATH, exist_ok=True)

    def _save(self, filename: str, data: dict):
        tmp = self.BASE_PATH + filename + ".tmp"
        final = self.BASE_PATH + filename
        with open(tmp, "w") as f:
            json.dump(data, f)
        os.replace(tmp, final)

    def _load(self, filename: str) -> dict:
        path = self.BASE_PATH + filename
        if not os.path.exists(path):
            return {}
        with open(path, "r") as f:
            return json.load(f)

    def save_properties(self, data: dict):
        self._save("properties.json", data)

    def load_properties(self):
        return self._load("properties.json")
"""
